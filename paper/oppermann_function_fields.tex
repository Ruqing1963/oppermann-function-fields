\begin{abstract}
For a monic $f\in\F_q[t]$ of degree $d$ the Legendre interval $\{f^2+s:\deg s\le d\}$ splits into the $q$
\emph{Oppermann sub-intervals} $O_{f,c}=\{f^2+cf+s:\deg s<d\}$, $c\in\F_q$, each of size $q^d$; for $q=2$ these are
the two halves, the analogue of $(n^2,n^2+n)$ and $(n^2+n,(n+1)^2)$. We ask whether every $O_{f,c}$ contains an
irreducible polynomial. In characteristic~$2$ the answer is no, and the failures are explained completely by three
Galois-theoretic obstructions: (A)~for $d=1$ the piece $c=0$ consists of squares; (B)~for $d=2$ we compute the exact
number of irreducibles in every Oppermann sub-interval, $q^2/4$, $0$, $q(q-2)/4$ or $q(q-1)/2$, by letting Frobenius act
on the roots of an additive polynomial through $\AGL_2(\F_2)\cong S_4$; it vanishes exactly when $q=4^j$, $a_1^2+c=0$, or
$q=2$, $a_1^2+c=1$; (C)~for $q=2$, $d=4$ the block $\{t^8+s:\deg s\le3\}$ contains no irreducible, by the
Stickelberger--Swan sign of Frobenius, because $8\mid\deg$ makes the sign depend only on the tail. In odd
characteristic every class of the Hayes group of level $d$ is an Oppermann sub-interval. For $d=2$ we prove the exact
count $\pi=(q^2-1-(q-\eta(-1))\eta(2e))/4>0$ for every odd $q$, via a ternary trace form on $\F_{q^4}$; with (B),
this settles $d=2$ for all $q$. An exact computation
with the Hayes explicit formula shows that all of them contain irreducibles for $q=3$, $d\le16$; $q=5$, $d\le10$;
$q=7$, $d\le9$; $q=9$, $d\le8$; $q=11$, $d\le7$; $q=13$, $d\le6$. In characteristic~$2$ the same computation, for
$q\in\{2,4,8,16\}$ up to $q^d\le2^{26}$, finds no empty Oppermann sub-interval beyond (A)--(C). The Weil bound sits
at the square-root barrier for these intervals. For $d=2$ it gives $\Psi\ge q$, and in the empty classes of (B)
this is attained exactly, by fourth powers of linear polynomials. For fixed $d$ and large $q$, the theorem of Bank,
Bary-Soroker and Rosenzweig gives nonemptiness in most cases; we determine which.
\end{abstract}

\section{Introduction}

Oppermann's conjecture \cite{Oppermann} asserts that for every $n\ge2$ both $(n^2,n^2+n)$ and $(n^2+n,(n+1)^2)$
contain a prime. It implies Legendre's conjecture and lies beyond the Riemann hypothesis, which gives primes only
in intervals $[x,x+C\sqrt x\log x]$. Over $\F_q[t]$, Legendre intervals $I_f=\{f^2+s:\deg s\le d\}$ ($f$ monic of
degree $d$) were studied in \cite{ChenI}. Since $f^2+cf$ and $f^2+ct^d$ differ by a polynomial of degree $<d$, the
interval $I_f$ is the disjoint union of the $q$ sub-intervals
\begin{equation}\label{eq:Ofc}
O_{f,c}=\{f^2+cf+s:\deg s<d\}=\{f^2+ct^d+s:\deg s<d\},\qquad c\in\F_q,
\end{equation}
each of cardinality $q^d=|f|$, the norm of $f$, exactly as $(n^2,n^2+n)$ has length $n$. For $q=2$ there are two
halves. We call $O_{f,c}$ the \emph{Oppermann sub-intervals} and ask:

\begin{quote}
\emph{Does every $O_{f,c}$ contain an irreducible polynomial?}
\end{quote}

\noindent The answer depends on the characteristic.

\begin{theorem}[characteristic $2$]\label{thm:main2}
Let $q=2^k$.
\begin{enumerate}
\item[(A)] For $d=1$ the sub-interval $O_{f,0}=\{t^2+s_0:s_0\in\F_q\}$, which is the same set for all $q$
polynomials $f$, consists of squares. Every $O_{f,c}$ with $c\ne0$ contains exactly $q/2$ irreducibles.
\item[(B)] For $d=2$ and $f=t^2+a_1t+a_0$, the number of irreducibles in $O_{f,c}$ depends only on $e=a_1^2+c$ and
equals
\[
\begin{array}{c|cc}
 & e=0 & e\ne0\\\hline
k\text{ even} & 0 & q^2/4\\
k\text{ odd} & q(q-1)/2 & q(q-2)/4
\end{array}
\]
The table has two zero entries: $k$ even with $e=0$, and $q(q-2)/4$ at $q=2$ (where $k=1$ is odd). Hence
$O_{f,c}$ is empty if and only if either $k$ is even and $c=a_1^2$, or $q=2$ and $c=a_1^2+1$. For $q=2$ the block
$e=0$ is \emph{not} empty: it contains $q(q-1)/2=1$ irreducible, $t^4+t+1$. For every
$q\in\{2,4,16,64,\dots\}$ there is thus exactly one empty sub-interval as a set: $B_0$, or $B_1$ when $q=2$. It is
represented by $q^2$ pairs $(f,c)$, one $c$ for each of the $q^2$ polynomials $f$. For $q\in\{8,32,128,\dots\}$
there is none.
\item[(C)] For $q=2$ and $d=4$ the block $Z_4=\{t^8+s:\deg s\le3\}$ contains no irreducible polynomial. It equals
$O_{f,c}$ for the $8$ pairs $f=t^4+a_2t^2+a_1t+a_0$ (that is, $a_3=0$), $c=a_2$.
\end{enumerate}
\end{theorem}

\begin{theorem}[odd characteristic, computer-assisted]\label{thm:odd}
Let $q$ be odd. Every Oppermann sub-interval $O_{f,c}$ contains an irreducible polynomial whenever
\[
\begin{gathered}q=3,\ d\le16;\qquad q=5,\ d\le10;\qquad q=7,\ d\le9;\\ q=9,\ d\le8;\qquad q=11,\ d\le7;\qquad q=13,\ d\le6.\end{gathered}
\]
Equivalently (Lemma~\ref{lem:odd}), every interval $\{P+s:\deg s<d\}$ with $P$ monic of degree $2d$ contains an
irreducible polynomial in these ranges.
\end{theorem}

\begin{theorem}[characteristic $2$, computer-assisted]\label{thm:even}
For $q=2$, $d\le26$; $q=4$, $d\le13$; $q=8$, $d\le7$; $q=16$, $d\le5$, the only empty Oppermann sub-intervals are
those of Theorem~\ref{thm:main2}.
\end{theorem}

\begin{theorem}[odd characteristic, $d=2$]\label{thm:odd2}
Let $q$ be odd and let $\eta$ be the quadratic character of $\F_q$, with $\eta(0)=0$. For $f=t^2+a_1t+a_0$ and
$c\in\F_q$, put $e=2a_0+c-a_1^2/2$. Then
\[
\pi(O_{f,c})=\frac{q^2-1-\bigl(q-\eta(-1)\bigr)\,\eta(2e)}{4}.
\]
In particular, every Oppermann sub-interval with $d=2$ contains at least $q(q-1)/4$ irreducibles if
$q\equiv1\pmod4$, and at least $(q+1)(q-2)/4$ if $q\equiv3\pmod4$. Together with Theorem~\ref{thm:main2}(B), this
settles $d=2$ for every $q$.
\end{theorem}

For fixed $d$ and large $q$, the theorem of Bank, Bary-Soroker and Rosenzweig \cite{BBR} settles the question in most
cases (Corollary~\ref{cor:BBR}). The tables above therefore matter mainly for small $q$, where that theorem gives
nothing.

Throughout, a \emph{sub-interval} means the set $O_{f,c}$. Distinct pairs $(f,c)$ can give the same set; counts of
sub-intervals are counts of sets, that is, of Hayes classes.

Section~\ref{sec:classes} identifies the Oppermann sub-intervals with Hayes classes and explains why
characteristic~$2$ is special: the Oppermann classes are concentrated on prefixes close to squares.
Section~\ref{sec:char2} proves Theorem~\ref{thm:main2}, and Section~\ref{sec:odd2} proves Theorem~\ref{thm:odd2}.
Section~\ref{sec:comp} describes the exact computation behind
Theorems~\ref{thm:odd} and~\ref{thm:even}. Section~\ref{sec:barrier} explains why the explicit formula alone cannot
go further.

\section{Oppermann sub-intervals as Hayes classes}\label{sec:classes}

For $P$ monic of degree $n$ write $P=t^n+p_{n-1}t^{n-1}+\dots+p_0$. Two monic polynomials of degree $n$ lie in the
same \emph{Hayes class of level $l$} if they agree in $p_{n-1},\dots,p_{n-l}$. Each class is a coset
$P+\{s:\deg s<n-l\}$, and the map
\[
P\mapsto u^nP(1/u)\bmod u^{l+1}=1+p_{n-1}u+\dots+p_{n-l}u^l
\]
identifies the classes with the elements of the multiplicative group $G_l=(1+u\F_q[u])/(1+u^{l+1}\F_q[u])$ of order
$q^l$. This map is multiplicative: $\mathrm{class}(PQ)=\mathrm{class}(P)\,\mathrm{class}(Q)$ in $G_l$. The characters
of $G_l$ are the short-interval characters of Hayes \cite{Hayes} and Keating--Rudnick \cite{KR}. A character has
\emph{exact level} $j$ if it factors through $G_j$ but not through $G_{j-1}$. There are $q^j-q^{j-1}$ characters of
exact level $j$, for $1\le j\le l$. By
\eqref{eq:Ofc}, $O_{f,c}$ is the Hayes class of level $d$ and degree $n=2d$ containing $f^2+ct^d$.

\begin{lemma}[odd $q$]\label{lem:odd}
If $q$ is odd, every Hayes class of level $d$ in degree $2d$ is an Oppermann sub-interval, and it equals $O_{f,c}$ for
exactly $q$ pairs $(f,c)$.
\end{lemma}

\begin{proof}
Write $f=\sum_{i=0}^da_it^i$ with $a_d=1$. For $1\le j\le d$ the coefficient of $t^{2d-j}$ in $f^2$ is
\[
[f^2]_{2d-j}=\sum_{\substack{i+i'=2d-j\\ 0\le i,i'\le d}}a_ia_{i'}
=2a_{d-j}+\Phi_j,\qquad
\Phi_j=\sum_{\substack{i+i'=2d-j\\ d-j<i,\,i'<d}}a_ia_{i'} ,
\]
where the term $2a_{d-j}=2a_da_{d-j}$ comes from $\{i,i'\}=\{d,d-j\}$. The sum $\Phi_j$ runs over ordered pairs, so
it includes the square $a_{d-j/2}^2$ when $j$ is even. $\Phi_j$ depends only on $a_{d-1},\dots,a_{d-j+1}$; for
example $\Phi_1=0$, $\Phi_2=a_{d-1}^2$, $\Phi_3=2a_{d-1}a_{d-2}$.

The map $(a_{d-1},\dots,a_1)\mapsto([f^2]_{2d-1},\dots,[f^2]_{d+1})$ is therefore triangular with diagonal entries
$2$. Since $2\in\F_q^\times$, it is a bijection $\F_q^{d-1}\to\F_q^{d-1}$: the prefix of a class determines
$a_{d-1},\dots,a_1$ uniquely, by solving $a_{d-j}=([f^2]_{2d-j}-\Phi_j)/2$ for $j=1,\dots,d-1$ in turn.

The remaining coefficient of the class is that of $t^d$ in $f^2+ct^d$, namely $2a_0+\Phi_d+c$, where $\Phi_d$ is
already fixed. For each of the $q$ values of $a_0$ there is exactly one $c$ giving the prescribed value. So every
class arises, and from exactly $q$ of the $q^{d+1}$ pairs $(f,c)$.
\end{proof}

So in odd characteristic the Oppermann question at degree $2d$ is the question whether \emph{every} short interval of
length $q^d=q^{n/2}$ contains a prime: no class is singled out.

\begin{lemma}[$q$ even]\label{lem:even}
Let $q$ be even. The Oppermann sub-intervals are exactly the Hayes classes of level $d$ (in degree $2d$) in which the
coefficient of $t^j$ is $0$ for every odd $j$ with $d<j<2d$. The coefficients of $t^d$ and of $t^j$ for even $j$ with
$d<j<2d$ are arbitrary. There are $q^{\lceil d/2\rceil}$ such classes among the $q^d$ classes of level $d$, and each
of them equals $O_{f,c}$ for exactly $q^{\lfloor d/2\rfloor+1}$ pairs $(f,c)$.
\end{lemma}

\begin{proof}
In characteristic $2$, $f^2=\sum a_i^2t^{2i}$ has no odd terms, $a\mapsto a^2$ is bijective on $\F_q$, and the
coefficient of $t^d$ is shifted by $c$. This identifies the classes that occur.

For the count, note that $a\mapsto a^2$ is additive. So the map $(f,c)\mapsto\mathrm{prefix}(f^2+ct^d)$ is a
surjective homomorphism of additive groups from $\F_q^{d+1}$, the coefficients $(a_0,\dots,a_{d-1},c)$, onto
$\F_q^{\lceil d/2\rceil}$. Its fibres therefore all have size $q^{d+1-\lceil d/2\rceil}=q^{\lfloor d/2\rfloor+1}$.
\end{proof}

For example, the empty classes of Theorem~\ref{thm:main2} at $d=1$, $2$ and $4$ are represented by $q$, $q^2$ and
$2^3=8$ pairs $(f,c)$ respectively.

Thus in characteristic $2$ the Oppermann sub-intervals are the short intervals around the $q^{\lceil d/2\rceil}$
``near-squares'' $f^2+ct^d$, a set of measure $q^{-\lfloor d/2\rfloor}$ among all short intervals. Squares and
near-squares carry Galois-theoretic structure, and that structure produces the counterexamples below.

\section{\texorpdfstring{Characteristic $2$}{Characteristic 2}: the three obstructions}\label{sec:char2}

Throughout this section $q=2^k$ and $\Tr=\Tr_{\F_q/\F_2}$.

\subsection{\texorpdfstring{(A) $d=1$}{(A) d=1}}
Here $f=t+a_0$ and $O_{f,c}=\{t^2+ct+s_0:s_0\in\F_q\}$.

For $c=0$ the members $t^2+s_0=(t+\sqrt{s_0})^2$ are squares.

For $c\ne0$, substitute $t=cy$: $t^2+ct+s_0=c^2(y^2+y+s_0/c^2)$. This is irreducible if and only if
$y^2+y+s_0/c^2$ has no root in $\F_q$, that is, if and only if $\Tr(s_0/c^2)=1$. Exactly $q/2$ values of $s_0$
satisfy this.

\subsection{\texorpdfstring{(B) $d=2$: an additive polynomial and $\AGL_2(\F_2)$}{(B) d=2: an additive polynomial and AGL(2,2)}}
For $f=t^2+a_1t+a_0$ we have $f^2=t^4+a_1^2t^2+a_0^2$, so
\[
O_{f,c}=B_e:=\{t^4+et^2+\alpha t+\beta:\ \alpha,\beta\in\F_q\},\qquad e=a_1^2+c .
\]
By Lemma~\ref{lem:even} the blocks $B_e$, $e\in\F_q$, are precisely the Oppermann classes for $d=2$.

\begin{proposition}\label{prop:agl}
Let $\alpha\ne0$. Put $L(t)=t^4+et^2+\alpha t$ and $R(y)=y^3+ey+\alpha$.
\begin{enumerate}
\item $t^4+et^2+\alpha t+\beta$ is irreducible for some $\beta$ if and only if $R$ has exactly one root in $\F_q$.
\item If $R$ has exactly one root in $\F_q$, then $t^4+et^2+\alpha t+\beta$ is irreducible for exactly $q/2$ values of
$\beta\in\F_q$.
\end{enumerate}
If $\alpha=0$, every member $t^4+et^2+\beta=(t^2+\sqrt e\,t+\sqrt\beta)^2$ is a square.
\end{proposition}

\begin{proof}
$L$ is $\F_2$-linear and separable ($L'=\alpha\ne0$), so $V=\ker L\subset\overline{\F}_q$ is a $2$-dimensional
$\F_2$-space, and $V\setminus\{0\}$ is the set of roots of $L(t)/t=R(t)$. The Frobenius $\varphi(x)=x^q$ acts on $V$ by
some $F\in\GL(V)\cong S_3$. Exactly one of three cases holds: $F=1$ ($R$ has three roots in $\F_q$), $F$ of order $3$
(no root), or $F$ a transvection (exactly one root).

Fix $\beta$ and a root $r$ of $P_\beta=L+\beta$. The roots of $P_\beta$ form the coset $r+V$. Since
$L(\varphi(r))=\varphi(\beta)=\beta$, we have $\varphi(r)=r+v$ with $v\in V$, and
$\varphi(r+w)=r+(Fw+v)$. So Frobenius acts on the roots through the affine map $T(w)=Fw+v$ of
$V\cong\F_2^2$, an element of $\AGL_2(\F_2)\cong S_4$. $P_\beta$ is irreducible if and only if $T$ is a $4$-cycle.
\begin{itemize}
\item If $F=1$, $T$ is a translation and $T^2=1$.
\item If $F$ has order $3$, $F-1$ is invertible, so $T$ has a fixed point.
\item If $F$ is a transvection, $T^2(w)=w+(F+1)v$, so $T$ is a $4$-cycle if and only if $(F+1)v\ne0$, i.e.
$v\notin V^F$.
\end{itemize}
This proves the ``only if'' part of (1); the ``if'' part follows from (2), since $q/2\ge1$. For (2), let $F$ be a
transvection, so $(F+1)V=V^F$ is a line. Replacing $r$ by $r+w$ replaces $v$ by
$v+(F+1)w$, so the class $[v]\in V/V^F\cong\F_2$ depends only on $\beta$, and $\beta\mapsto[v]$ is additive. Its
kernel consists of the $\beta$ for which some root $r'=r+w$ is fixed by $\varphi$, i.e.\ $\beta\in L(\F_q)$. The map
$L:\F_q\to\F_q$ is $\F_2$-linear with kernel $V^F=V\cap\F_q$ of order $2$, so $|L(\F_q)|=q/2$ and exactly $q/2$ values
of $\beta$ have $[v]\ne0$.
\end{proof}

\begin{lemma}\label{lem:oneroot}
Let $\alpha\ne0$. The cubic $R(y)=y^3+ey+\alpha$ has exactly one root in $\F_q$ if and only if
$\Tr(e^3/\alpha^2)\ne\Tr(1)$.
\end{lemma}

\begin{proof}
The roots $y_1,y_2,y_3$ of $R$ are distinct ($R$ is separable as above), with $y_1+y_2+y_3=0$ and $y_1y_2y_3=\alpha$.
Put $\rho=y_1/y_3+y_2/y_1+y_3/y_2$ and $\rho'=y_2/y_3+y_1/y_2+y_3/y_1$. Even permutations of the roots fix $\rho$ and
odd ones exchange $\rho,\rho'$; using $y_i+y_j=y_k$,
\[
\rho+\rho'=\frac{y_1+y_2}{y_3}+\frac{y_2+y_3}{y_1}+\frac{y_3+y_1}{y_2}=1 .
\]
Hence $\rho,\rho'$ are the roots of $z^2+z+\delta$ with $\delta=\rho\rho'\in\F_q$, and Frobenius acts on the roots of
$R$ by an odd permutation (a transposition: exactly one root in $\F_q$) if and only if $\rho\notin\F_q$, i.e.\
$\Tr(\delta)=1$. Expanding,
\[
\delta=3+\sum_k\frac{y_iy_j}{y_k^2}+\sum_k\frac{y_k^2}{y_iy_j}
=1+\sum_k\frac{\alpha}{y_k^3}+\frac{1}{\alpha}\sum_k y_k^3 .
\]
From $y_k^3=ey_k+\alpha$ we get $\sum_k y_k^3=\alpha$, so $\delta=\sum_k\alpha/w_k$ with $w_k=y_k^3=ey_k+\alpha$.
Substituting $y=(w+\alpha)/e$ (for $e\ne0$) into $w=y^3$ shows that the $w_k$ are the roots of
$w^3+\alpha w^2+(\alpha^2+e^3)w+\alpha^3$, so $\sum_k1/w_k=(\alpha^2+e^3)/\alpha^3$ and
\[
\delta=1+e^3/\alpha^2 .
\]
For $e=0$ this also holds directly ($w_k=\alpha$). So $\Tr(\delta)=1$ if and only if $\Tr(e^3/\alpha^2)\ne\Tr(1)$.
\end{proof}

\begin{remark}
$\delta$ is Berlekamp's characteristic-$2$ discriminant \cite{Berlekamp76} of $R$, and $R$ is the cubic resolvent of
the quartic; the proof above is self-contained. For $e=0$ the lemma says: $y^3=\alpha$ has exactly one root iff $k$ is
odd, which is the familiar fact that cubing is bijective on $\F_q$ iff $3\nmid q-1$.
\end{remark}

\begin{proof}[Proof of Theorem~\ref{thm:main2}(B)]
By Proposition~\ref{prop:agl}, $|B_e\cap\{\text{irreducibles}\}|=\frac q2\cdot\#\{\alpha\ne0:\Tr(e^3/\alpha^2)\ne\Tr(1)\}$.
If $e=0$ the condition reads $\Tr(1)=1$, i.e.\ $k$ odd: the count is $\frac q2(q-1)$ for $k$ odd and $0$ for $k$ even.
If $e\ne0$, $\alpha\mapsto e^3/\alpha^2$ permutes $\F_q^\times$, and $\#\{x\ne0:\Tr(x)=1\}=q/2$,
$\#\{x\ne0:\Tr(x)=0\}=q/2-1$; with $\Tr(1)=k\bmod2$ this gives $q^2/4$ ($k$ even) and $\frac q2(\frac q2-1)$ ($k$
odd). The count $q(q-2)/4$ vanishes only for $q=2$.
\end{proof}

The formula was checked by brute force for $q\le64$ (script \texttt{char2\_d2\_theorem.py}): criterion of
Lemma~\ref{lem:oneroot}, the $q/2$ count of Proposition~\ref{prop:agl}(2), and the four block totals.

\subsection{\texorpdfstring{(C) $q=2$, $d=4$: the sign of Frobenius}{(C) q=2, d=4: the sign of Frobenius}}

The Oppermann classes for $q=2$, $d=4$ are the four prefixes $(p_7,p_6,p_5,p_4)=(0,x,0,y)$ (Lemma~\ref{lem:even}).
For $x=y=0$ the class is the \emph{zero block}
\[
Z_4=\{t^8+a_3t^3+a_2t^2+a_1t+a_0\}\subset\F_2[t],
\]
equal to $O_{f,c}$ for $f=t^4+a_2t^2+a_1t+a_0$ and $c=a_2$. Here $a_3=0$ is forced, because the coefficient of
$t^6$ in $f^2$ is $a_3^2$; this is why there are $2^3=8$ pairs rather than $16$.

We use the theorem of Stickelberger \cite{Stickelberger} in the form of Swan \cite[Cor.~1]{Swan}: \emph{if
$\tilde P\in\Z[t]$ is monic of degree $n$ and its reduction $P\in\F_2[t]$ is squarefree with $r$ irreducible
factors, then $r\equiv n\pmod2$ if and only if $\disc(\tilde P)\equiv1\pmod8$.} Equivalently, the Frobenius
permutation of the roots of $P$ is even if and only if $\disc(\tilde P)\equiv1\pmod 8$. An irreducible $P$ of degree $8$
has Frobenius an $8$-cycle, an odd permutation, and would need $\disc(\tilde P)\equiv5\pmod8$.

\begin{lemma}\label{lem:tail}
Let $\tilde P=t^n+\tilde g$ with $\tilde g\in\Z[t]$, $\deg\tilde g<n$, and $8\mid n$. Then
$\disc(\tilde P)\equiv\Res(\tilde P,\tilde g')\pmod8$.
\end{lemma}

\begin{proof}
$\disc(\tilde P)=(-1)^{n(n-1)/2}\Res(\tilde P,\tilde P')=\prod_{\tilde P(\rho)=0}(n\rho^{n-1}+\tilde g'(\rho))$, the
sign being $+1$ for $8\mid n$. Expanding the product, every term containing a factor $n$ is $n$ times a symmetric
polynomial with integer coefficients in the roots, hence lies in $8\Z$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main2}(C)]
An irreducible member of $Z_4$ has no root in $\F_2$, so $a_0=1$ and $a_1+a_2+a_3=1$. This leaves four polynomials.
$t^8+t^2+1=(t^4+t+1)^2$ is a square. For the other three choose integer lifts and apply Lemma~\ref{lem:tail}:
\begin{itemize}
\item $t^8+t+1$: $\tilde g'=1$, $\disc\equiv1$;
\item $t^8+t^3+1$: $\tilde g'=3t^2$, $\Res(\tilde P,3t^2)=3^8\big(\prod\rho\big)^2=3^8\equiv1$;
\item $t^8+t^3+t^2+t+1$, lifted as $t^8+(t+1)^3$: $\tilde g'=3(t+1)^2$, and
$\Res(\tilde P,3(t+1)^2)=3^8\prod(\rho+1)^2=3^8\tilde P(-1)^2$, with $\tilde P(-1)=(-1)^8+0^3=1$, so
$\disc\equiv3^8\equiv1$.
\end{itemize}
Suppose one of these three were irreducible over $\F_2$. Then it would be squarefree with $r=1$, and
Stickelberger--Swan would require $\disc(\tilde P)\equiv5\pmod8$. That contradicts $\disc(\tilde P)\equiv1$, so none of
them is irreducible. The argument does not need squarefreeness to be checked beforehand.
\end{proof}

\begin{remark}
The mechanism is that $8\mid\deg P$ kills the leading term of $P'$ modulo $8$, so the sign of Frobenius is decided by
the tail $g$ alone; with $\deg g\le3$ the tail is too short to produce the sign $-1$ on a polynomial with no linear
factor. Table~\ref{tab:zero} shows the zero blocks $Z_d=\{t^{2d}+s:\deg s<d\}$, which are Oppermann classes for every
$d$. For $d=8$ ($8\mid 2d$ again) the tail has degree $\le7$, half of the squarefree members already have the sign
$-1$, and the block contains $10$ irreducibles. The parallel with (B) is exact: in (B) the obstruction is the
quotient $S_4\to S_3$ (the resolvent cubic), in (C) the quotient $S_8\to\{\pm1\}$ (the sign).
\end{remark}

\section{\texorpdfstring{Odd characteristic, $d=2$: a ternary trace form}{Odd characteristic, d=2: a ternary trace form}}\label{sec:odd2}

Throughout this section $q$ is odd and $\eta$ is the quadratic character of $\F_q$, with $\eta(0)=0$.

\subsection{Reduction to one-parameter blocks}
The level-$2$ classes in degree $4$ are $C_{p_3,p_2}=\{t^4+p_3t^3+p_2t^2+\alpha t+\beta:\alpha,\beta\in\F_q\}$. The
translation $t\mapsto t-p_3/4$ is defined because $4$ is invertible (in characteristic $3$ it is $t\mapsto t-p_3$).
It preserves irreducibility and maps $C_{p_3,p_2}$ bijectively onto
\[
B_e=\{t^4+et^2+\alpha t+\beta:\alpha,\beta\in\F_q\},\qquad e=p_2-\tfrac38p_3^2 .
\]
For $O_{f,c}$ with $f=t^2+a_1t+a_0$ we have $p_3=2a_1$ and $p_2=a_1^2+2a_0+c$, so $e=2a_0+c-a_1^2/2$.

\subsection{The count}
\begin{proposition}\label{prop:odd2}
For every $e\in\F_q$, the number of irreducibles in $B_e$ is
\[
\pi(B_e)=\frac{q^2-1-\bigl(q-\eta(-1)\bigr)\eta(2e)}{4}.
\]
\end{proposition}

\begin{proof}
A monic irreducible quartic $P\in B_e$ is the minimal polynomial of exactly four elements $x\in\F_{q^4}$ of degree
$4$, namely its roots. Write $\Tr=\Tr_{\F_{q^4}/\F_q}$. The coefficient conditions on $P$ translate into conditions on
$x$: $p_3=-\Tr(x)$, and when $\Tr x=0$,
\[
p_2=e_2(x,x^q,x^{q^2},x^{q^3})=\tfrac12\bigl((\Tr x)^2-\Tr(x^2)\bigr)=-\tfrac12\Tr(x^2).
\]
Hence $4\pi(B_e)=N-M$, where
\[
N=\#\{x\in W:Q(x)=-2e\},\qquad W=\ker\Tr,\quad Q(x)=\Tr(x^2),
\]
and $M$ counts the elements of $W$ with $Q(x)=-2e$ that have degree $<4$ over $\F_q$, i.e.\ that lie in
$\F_{q^2}$.

\emph{The form $Q$.} The bilinear form $(x,y)\mapsto\Tr(xy)$ on $\F_{q^4}$ is nondegenerate. Its Gram determinant on
a basis $\omega_1,\dots,\omega_4$ is $D=\det(\omega_i^{q^{j}})^2$. Frobenius permutes the columns of the matrix
$(\omega_i^{q^j})$ by a $4$-cycle, an odd permutation, so $\det(\omega_i^{q^j})$ is not fixed by Frobenius. It
therefore lies outside $\F_q$ while its square $D$ lies in $\F_q$, so $D$ is a nonsquare in $\F_q$.

Since $\Tr(1\cdot x)=\Tr x$, the line $\F_q$ is orthogonal to $W$, and $\F_{q^4}=\F_q\perp W$. The Gram determinant
of $\F_q$ is $\Tr(1)=4$, a square. So $Q|_W$ is a nondegenerate ternary form whose determinant $\Delta$ is a
nonsquare.

\emph{Counting $N$.} For a nondegenerate quadratic form in $3$ variables over $\F_q$ with determinant $\Delta$, the
number of solutions of $Q(x)=b$ is $q^2+q\,\eta(-b\Delta)$ \cite[Thm.~6.27]{LN}; for $b=0$ this is $q^2$. With
$b=-2e$ and $\eta(\Delta)=-1$ we get $N=q^2-q\,\eta(2e)$, also for $e=0$.

\emph{Counting $M$.} An element of $\F_{q^2}$ has $\Tr x=2\Tr_{\F_{q^2}/\F_q}(x)$, so it lies in $W$ if and only if
$x^q=-x$. Such $x$ form the line $\delta\F_q$, where $\delta^2=\nu$ is a nonsquare of $\F_q$. For these $x$,
$x^2\in\F_q$ and $Q(x)=2(x^2+x^{2q})=4x^2$, so the condition is $x^2=-e/2$. The squares of elements $s\delta$ are
$0$ and the nonsquares of $\F_q$, each nonsquare attained twice. Hence $M=1$ if $e=0$, and $M=1-\eta(-e/2)$ if
$e\ne0$; in both cases $M=1-\eta(-2e)$.

Therefore
\[
4\pi(B_e)=q^2-q\,\eta(2e)-1+\eta(-2e)=q^2-1-\bigl(q-\eta(-1)\bigr)\eta(2e). \qedhere
\]
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:odd2}]
Combine the reduction with Proposition~\ref{prop:odd2}. The minimum over $e$ is attained at $\eta(2e)=1$. It equals
$(q^2-1-(q-1))/4=q(q-1)/4$ when $\eta(-1)=1$, i.e.\ $q\equiv1\pmod4$, and $(q^2-1-(q+1))/4=(q+1)(q-2)/4$ when
$q\equiv3\pmod4$. Both are positive for every odd $q$.
\end{proof}

\begin{remark}
The values agree with a brute-force count of all $q^2$ members of every block $B_e$ for
$q=3,5,7,9,11,13,25,27,49,81$, and, for $q\le13$, of every class $C_{p_3,p_2}$; the script is
\texttt{odd\_d2\_formula.py}. They also agree with the minima of Table~\ref{tab:odd}. Characteristic $3$ needs no
special treatment.

The contrast with characteristic $2$ is instructive. There the translation $t\mapsto t+a$ cannot remove $p_3$,
since $4=0$. The Oppermann blocks are exactly the classes with $p_3=0$, on which $t^4+et^2+\alpha t$ is
\emph{additive}, and this additivity produces the obstruction of Proposition~\ref{prop:agl}. In odd characteristic
the trace form $\Tr(x^2)$ has the same shape for every $e$, and the count never drops below about $q^2/4-q/4$.
\end{remark}

\section{The exact computation}\label{sec:comp}

\subsection{Method}
Here $\Lambda$ is the von Mangoldt function: $\Lambda(F)=\deg\varpi$ if $F=\varpi^k$ for a monic irreducible
$\varpi$ and some $k\ge1$, and $\Lambda(F)=0$ otherwise. For a class $x$ of level $l$ and degree $n$, let
$\Psi(x)=\sum_{F\in x}\Lambda(F)$. The explicit formula
\cite{Hayes,KR} reads
\begin{equation}\label{eq:explicit}
\Psi(x)=q^{n-l}+q^{-l}\sum_{\chi\ne1}\overline{\chi}(x)\,\psi_n(\chi),\qquad
u\frac{L'(u,\chi)}{L(u,\chi)}=\sum_{m\ge1}\psi_m(\chi)u^m,
\end{equation}
where for $\chi\ne1$ the $L$-function $L(u,\chi)=\sum_{F}\chi(F)u^{\deg F}$ is a polynomial of degree $\le l-1$. Its
coefficients are the Fourier transforms on $G_l$ of the class distributions of the monic polynomials of degree $<l$;
$\psi_n(\chi)$ follows by the Newton recursion, and one inverse FFT on $G_l$ returns $\Psi(x)$ for all $q^l$ classes
at once. This is the algorithm of \cite{ChenI}, run here at level $l=d$ and degree $n=2d$. The number of irreducibles
in $x$ is
\[
\pi(x)=\frac1n\bigl(\Psi(x)-E(x)\bigr),\qquad
E(x)=\sum_{k\ge2,\ k\mid n}\ \sum_{\deg\varpi=n/k,\ \varpi^k\in x}\frac nk .
\]
The proper prime powers in $E(x)$ are enumerated directly (Ben-Or test in degree $\le d$) and placed by the rule
$\mathrm{class}(\varpi^k)=\mathrm{class}(\varpi)^k$ in $G_l$. Each run checks that every $\pi(x)$ is
an integer and that $\sum_x(n\pi(x)+E(x))=q^n$, and reports the largest distance of the floating-point $\Psi(x)$ from
an integer (below $10^{-7}$ in all runs). As an independent check, a brute-force census that tests every polynomial
of degree $2d$ for irreducibility (\texttt{verify\_pointwise\_oppermann.py}) agrees with the FFT counts class by
class for all $q^{2d}\le2^{22}$ (\texttt{compare\_bruteforce.py}), and in minimum, mean and empty classes for all
$q^{2d}\le2^{26}$.

\section{The square-root barrier}\label{sec:barrier}

In \eqref{eq:explicit} with $n=2d$ and $l=d$ the main term is $q^d$.

Let $\chi$ have exact level $j$. The monic polynomials of degree $\ge j$ are equidistributed over $G_j$, so
$\deg L(u,\chi)\le j-1$. By Weil, every inverse zero $\gamma$ of $L(u,\chi)$ has $|\gamma|\in\{1,\sqrt q\}$, hence
$|\psi_{2d}(\chi)|\le(j-1)q^d$. Characters of level $1$ have $L=1$ and contribute nothing. Multiplying by the factor
$q^{-d}$ in \eqref{eq:explicit}, which cancels the $q^d$, and summing over the $q^j-q^{j-1}$ characters of each
exact level $j$ gives
\begin{equation}\label{eq:weil}
|\Psi(x)-q^d|\le\sum_{j=2}^{d}(j-1)(q^j-q^{j-1})=(d-1)q^d-\sum_{j=1}^{d-1}q^j .
\end{equation}

\emph{For $d\ge3$} the right-hand side exceeds $q^d$ for every $q$, so \eqref{eq:weil} does not even give
$\Psi(x)>0$. This is the exact analogue of the situation over $\Z$: Oppermann's intervals have length $\sqrt x$, and
the Riemann hypothesis falls short by a factor $\log x$. No degree threshold beyond which $O_{f,c}\ne\emptyset$
follows from Weil's theorem.

\emph{For $d=2$}, \eqref{eq:weil} reads $|\Psi(x)-q^2|\le q^2-q$, so $\Psi(x)\ge q>0$ for every class. This still
does not produce an irreducible, because $\Psi$ also counts prime powers. Theorem~\ref{thm:main2}(B) shows the
lower bound is sharp. Take $q=4^j$ and the empty class $B_0=\{t^4+\alpha t+\beta\}$. Its proper prime powers are
the $q$ fourth powers $(t+c_0)^4=t^4+c_0^4$, each with $\Lambda=1$; the squares $(t^2+b)^2$ do not count, since
$t^2+b$ is itself a square. Hence
\[
\Psi(B_0)=4\pi(B_0)+E(B_0)=0+q=q .
\]
So equality holds in \eqref{eq:weil}, which forces $\overline\chi(B_0)\psi_4(\chi)=-q^2$ for all $q^2-q$ characters
of exact level $2$. The whole of the Weil lower bound is taken up by the fourth powers.

The same happens for $q=2$ on $B_1$, where $\Psi(B_1)=E(B_1)=2$ comes from $(t^2+t+1)^2$. The values
$\Psi(B_0)=q$ for $q=4,16$ and $\Psi(B_1)=2$ for $q=2$ were confirmed with the FFT code.

Results on the variance of $\Psi$ over Legendre intervals (an average over $f$) bound the number of intervals that
deviate by a given amount; they do not exclude a single empty interval.

For fixed degree and $q\to\infty$, Bank, Bary-Soroker and Rosenzweig \cite{BBR} prove the following. For monic
$P\in\F_q[t]$ of degree $k$ and $1\le m<k$, the number $N$ of irreducibles in $\{P+h:\deg h\le m\}$ satisfies
\[
\bigl|N-q^{m+1}/k\bigr|\le c(k)\,q^{m+1/2},
\]
provided that $m\ge2$ if $\gcd(q,k(k-1))>1$, and $m\ge3$ if $q$ is even and $\deg P'\le1$. They also note that the
estimate fails outside this range.

Our blocks $B_e=\{t^4+et^2+h:\deg h\le1\}$ have $k=4$ and $m=1$, and in characteristic $2$ the centre
$P=t^4+et^2$ has $P'=0$, so both conditions exclude them. Theorem~\ref{thm:main2}(B) gives the exact count at this
excluded point and shows how the asymptotic fails:
\begin{itemize}
\item for $q=4^j$ the block $e=0$ contains no irreducible, while the expected count is $q^2/4$;
\item for $q=2^{2j+1}$ the block $e=0$ contains $q(q-1)/2$, about twice the expected count.
\end{itemize}
The blocks $e\ne0$, with $q^2/4$ and $q(q-2)/4$ irreducibles, do agree with the asymptotic. The obstruction comes
from the additive structure of $t^4+et^2+\alpha t$ (Proposition~\ref{prop:agl}). For $m=1$, $k=4$ and $q$ even,
the conditions of \cite{BBR} therefore cannot be dropped. By contrast, when $3\mid q$, a case that is also excluded by
$\gcd(q,k(k-1))>1$ at $k=4$, $m=1$, Theorem~\ref{thm:odd2} shows that the asymptotic $q^2/4+O(q)$ continues to hold
without exception.

Theorem~\ref{thm:main2}(C) has $q=2$ fixed, so it lies outside the $q\to\infty$ regime altogether.

In the other direction, \cite{BBR} gives nonemptiness for large $q$ in most cases. An Oppermann sub-interval
$O_{f,c}=\{f^2+ct^d+h:\deg h\le d-1\}$ has $k=2d$ and $m=d-1$. Since $N\ge q^d/2d-c(2d)q^{d-1/2}$, the
sub-interval is nonempty as soon as $q>(2d\,c(2d))^2$, provided the conditions of \cite{BBR} hold. We do not make the
constant $c(2d)$ explicit.

\begin{corollary}[of \cite{BBR}]\label{cor:BBR}
For each $d$ there is $q_0(d)$ such that every Oppermann sub-interval of degree $2d$ over $\F_q$, with $q\ge q_0(d)$,
contains an irreducible polynomial in each of the following cases:
\begin{enumerate}
\item $q$ odd and $d\ge3$;
\item $q$ odd, $d=2$ and $\gcd(q,6)=1$;
\item $q$ even and $d\ge4$;
\item $q$ even, $d=3$, and the class has $p_3\ne0$.
\end{enumerate}
\end{corollary}

\begin{proof}
We check the conditions of \cite{BBR} with $k=2d$ and $m=d-1$; recall that $1\le m<k$ is required.
\begin{itemize}
\item \emph{$q$ odd.} The only condition is $m\ge2$ when $p\mid 2d(2d-1)$. It holds for $d\ge3$. For $d=2$,
$k(k-1)=12$, so it holds when $\gcd(q,6)=1$.
\item \emph{$q$ even, $d\ge4$.} Here $m\ge3$, so both conditions hold.
\item \emph{$q$ even, $d=3$.} By Lemma~\ref{lem:even} every member $P$ of an Oppermann class has $p_5=0$. In
characteristic $2$ this gives $P'=p_3t^2+p_1$, so $\deg P'=2$ when $p_3\ne0$, and the condition ``$m\ge3$ if
$\deg P'\le1$'' does not apply.
\end{itemize}
\end{proof}

The cases left open by the corollary are the following. For $d=1$ the interval is too short for \cite{BBR}, but
Theorem~\ref{thm:main2}(A) and, for odd $q$, the count $(q-1)/2$ of irreducible $t^2+bt+s_0$ settle it. For $d=2$ the
corollary is superseded: Theorems~\ref{thm:main2}(B) and~\ref{thm:odd2} give exact counts for every $q$, including
$3\mid q$, with no threshold.
For $q$ even with $d=3$, the $q$ classes with $p_3=0$ (prefixes $(p_5,p_4,p_3)=(0,x,0)$) are treated in
\S\ref{subsec:p3zero}. In every case, $q<q_0(d)$ is covered only by computation.

\subsection{\texorpdfstring{The classes $p_3=0$ for $d=3$ in characteristic $2$}{The classes p3=0 for d=3 in characteristic 2}}\label{subsec:p3zero}
Let $q=2^k$ and
\[
C_e=\{t^6+et^4+s_2t^2+s_1t+s_0:\ s_2,s_1,s_0\in\F_q\},\qquad e\in\F_q .
\]
Every member has $P'=s_1$, so \cite{BBR} does not apply.

\begin{proposition}\label{prop:p3zero}
\begin{enumerate}
\item The number $\pi(C_e)$ does not depend on $e$.
\item $E(C_e)=q^2$; hence $\Psi(C_e)=6\pi(C_e)+q^2$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) In characteristic $2$, $(t+a)^6=(t^4+a^4)(t^2+a^2)=t^6+a^2t^4+a^4t^2+a^6$, and $(t+a)^4=t^4+a^4$. So
$t\mapsto t+a$ maps $C_e$ bijectively onto $C_{e+a^2}$ and preserves irreducibility. Since $a\mapsto a^2$ is
bijective, all $C_e$ are equivalent.

(2) We count the proper prime powers of degree $6$ in $C_0$.
\begin{itemize}
\item \emph{Sixth powers.} $(t+a)^6\in C_0$ only for $a=0$; this is $t^6$, with $\Lambda=1$.
\item \emph{Cubes.} For a quadratic $\varpi=t^2+bt+c$, the coefficient of $t^5$ in $\varpi^3$ is $b$. So $\varpi^3\in C_0$
forces $b=0$, but then $\varpi=(t+\sqrt c)^2$ is not irreducible.
\item \emph{Squares.} For a cubic $\varpi=t^3+a_2t^2+a_1t+a_0$ we have $\varpi^2=t^6+a_2^2t^4+a_1^2t^2+a_0^2$, which lies
in $C_0$ exactly when $a_2=0$. Translation changes $a_2$ by $3c=c$, so the $(q^3-q)/3$ irreducible cubics are
equidistributed over the values of $a_2$, and $(q^2-1)/3$ of them have $a_2=0$. Each contributes $\Lambda=3$.
\end{itemize}
Hence $E(C_0)=1+(q^2-1)=q^2$.
\end{proof}

\begin{conjecture}\label{conj:p3zero}
For $q=2^k$ and every $e\in\F_q$,
\[
\pi(C_e)=\begin{cases}q(q^2-1)/6, & k\text{ odd},\\ q(q-1)^2/6, & k\text{ even};\end{cases}
\qquad\text{equivalently}\qquad \Psi(C_e)=q^3-(-1)^k(q^2-q).
\]
In particular, $\pi(C_e)>0$ for every $q$.
\end{conjecture}

The conjectured formula holds for all $q\le256$; the counts are exhaustive (Table~\ref{tab:p3zero}, script
\texttt{char2\_d3\_p3zero.py}). Two pieces of evidence point towards a proof.

\begin{itemize}
\item For $q\le16$, every character $\chi$ of exact level $2$ or $3$ of $G_3$ has $L(u,\chi)$ with all inverse zeros
satisfying $\gamma^6\in q^3\{\pm1,\pm i\}$; these $L$-functions are supersingular.
\item By the Artin--Hasse factorisation $1+xu\equiv E(xu)E(x^2u^2)(1+x^3u^3)\pmod{u^4}$, the sums
$\psi_6(\chi)=\sum_{x\in\F_{q^6}}\chi(N(1+xu))$ become Gauss sums of $\Z/4$-valued quadratic forms on $\F_{q^6}$. Their
radicals are cut out by cubics $\gamma^2z^3+z+\gamma$ with $\gamma\in\F_q$, which split in $\F_{q^6}$.
\end{itemize}

A proof along these lines would need the Brown invariant of each of these forms; we do not carry this out here. Note the error term: it is $O(q)$ for $k$ odd and $-q^2/3+O(q)$ for
$k$ even, well inside the $O(q^{5/2})$ that a monodromy argument would give.



\section{Questions}
\begin{enumerate}
\item In characteristic $2$, are (A)--(C) the only empty Oppermann sub-intervals for all $q$ and $d$? By
Corollary~\ref{cor:BBR}, for each $d\ge4$ only finitely many $q$ can carry further empty sub-intervals. For $d=3$, the
classes with $p_3=0$ are not covered by \cite{BBR}. They all have the same count (Proposition~\ref{prop:p3zero}),
which is positive for $q\le256$; Conjecture~\ref{conj:p3zero} would settle them for all $q$.
\item For odd $q$, is every Oppermann sub-interval nonempty, for all $q$ and $d$? Lemma~\ref{lem:odd} turns
this into a statement about all short intervals of length $q^{n/2}$. By Corollary~\ref{cor:BBR}, for each $d\ge3$ it
fails for at most finitely many $q$. For $d=2$ it holds for every odd $q$ (Theorem~\ref{thm:odd2}). An explicit $q_0(d)$
combined with Theorem~\ref{thm:odd} would settle further small $d$ completely. The trace-form method of
Section~\ref{sec:odd2} would need, for $d=3$, the joint distribution of $\Tr(x^2)$ and $\Tr(x^3)$ on the subspace
$\ker\Tr\subset\F_{q^6}$; it does not reduce to a quadratic form.
\item Outside the characteristic-$2$ cases above, is there a constant $c>0$ with
$\min_x\pi(x)\ge c\,q^d/2d$ for all large $d$?
\end{enumerate}
